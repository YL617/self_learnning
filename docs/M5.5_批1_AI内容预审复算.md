def postorder(pre, ino):
    if not pre:
        return ''
    k = ino.index(pre[0])
    return postorder(pre[1:k+1], ino[:k]) + postorder(pre[k+1:], ino[k+1:]) + pre[0]

def preorder(post, ino):
    if not post:
        return ''
    k = ino.index(post[-1])
    return post[-1] + preorder(post[:k], ino[:k]) + preorder(post[k:-1], ino[k+1:])

def heap_sort(equal_descend):
    a = [(5, '5a'), (3, '3'), (5, '5b'), (2, '2')]
    def sift(i, size):
        while 2*i+1 < size:
            c = 2*i+1
            if c+1 < size and (a[c+1][0] > a[c][0] or
                              equal_descend and a[c+1][0] == a[c][0]):
                c += 1
            if a[i][0] > a[c][0] or (not equal_descend and a[i][0] == a[c][0]):
                break
            a[i], a[c] = a[c], a[i]
            i = c
    for i in range(len(a)//2-1, -1, -1):
        sift(i, len(a))
    for end in range(len(a)-1, 0, -1):
        a[0], a[end] = a[end], a[0]
        sift(0, end)
    return [label for _, label in a]
